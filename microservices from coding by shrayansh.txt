import org.springframework.web.bind.annotation.*;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;

@RestController
@RequestMapping("/orders")
public class OrderController {

    @GetMapping("/{id}")
    public String getOrder(@PathVariable String id) {

        HttpURLConnection httpURLConnection = null;

        try {
            // 1️⃣ Create URL
            String url = "http://localhost:8082/products/" + id;
            URL obj = new URL(url);

            // 2️⃣ Open Connection
            httpURLConnection = (HttpURLConnection) obj.openConnection();//here connection is not made its just the made of http object

            // 3️⃣ Set Request Method
            httpURLConnection.setRequestMethod("GET");

            // 4️⃣ Set Headers
            httpURLConnection.setRequestProperty("Accept", "application/json");

            // 5️⃣ Set Timeouts
            httpURLConnection.setConnectTimeout(5000);//means in how much time the connection should be build
            httpURLConnection.setReadTimeout(5000);mena sin how much the connection will be clsed if idle

            // 6️⃣ Get Response Code
           
            // 7️⃣ Read Response
            BufferedReader in = new BufferedReader(
                    new InputStreamReader(httpURLConnection.getInputStream())//connection made and request and response received at this stage//we can use here the httpURLConnection.getResponse code () and connect() too
            );

            String inputLine;
            StringBuilder response = new StringBuilder();

            while ((inputLine = in.readLine()) != null) {
                response.append(inputLine);
            }

            in.close();

            return "Order Service Response -> " + response.toString();

        } catch (Exception e) {
            return "Error: " + e.getMessage();
        } finally {
            if (httpURLConnection != null) {
                httpURLConnection.disconnect();
            }
        }
    }
}

import org.springframework.http.*;
import org.springframework.web.client.RestTemplate;

public class OrderServiceClient {

    public static void main(String[] args) {

        RestTemplate restTemplate = new RestTemplate();

        // URL
        String url = "http://localhost:8080/api/products/";

        // -------------------------
        // Customizing Headers
        // -------------------------
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_JSON);
        headers.set("Authorization", "Bearer my-token");

        // -------------------------
        // Preparing Request Body
        // -------------------------
        Product product = new Product();
        product.setName("Ice-cream");
        product.setPrice(100);

        // -------------------------
        // Setting Header + Body
        // -------------------------
        HttpEntity<Product> requestEntity =
                new HttpEntity<>(product, headers);

        // -------------------------
        // Calling exchange() method
        // -------------------------
        ResponseEntity<Product> response =
                restTemplate.exchange(
                        url,
                        HttpMethod.POST,
                        requestEntity,
                        Product.class
                );

        System.out.println("Response Status: " + response.getStatusCode());
        System.out.println("Response Body: " + response.getBody());
    }
}

*import org.springframework.cloud.client.ServiceInstance;
import org.springframework.cloud.client.discovery.DiscoveryClient;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.client.RestTemplate;

import java.net.URI;
import java.util.List;

public class ProductServiceCaller {

    @Autowired
    private DiscoveryClient discoveryClient;

    public void callProductAPI(String id) {

        RestTemplate restTemplate = new RestTemplate();

        // 1. Fetching the instances of "product-service"
        List<ServiceInstance> instances =
                discoveryClient.getInstances("product-service");

        // 2. Load balancer logic to choose a particular instance
        URI uri = instances.get(0).getUri();

        String response = restTemplate.getForObject(
                uri + "/products/" + id,
                String.class
        );

        System.out.println("Response from Product API call is: " + response);
    }
}
